Határozza meg az
\[
	f(x) = \frac{\sin(x^2 - 4)}{x^2 - 2x}
\]
függvény alábbi határértékeit, ha léteznek!
\begin{enumerate}
	\item \(\displaystyle \lim_{x \to -\infty} f(x);\)

	\item \(\displaystyle \lim_{x \to -2} f(x);\)

	\item \(\displaystyle \lim_{x \to 0} f(x);\)

	\item \(\displaystyle \lim_{x \to 2} f(x);\)
\end{enumerate}
