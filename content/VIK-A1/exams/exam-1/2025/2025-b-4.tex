\[
	f(x) = \frac{x \cdot \sin(x^2 - 9)}{x^2 + x - 6}
\]
\begin{enumerate}
	\item Határozza meg azt a legbővebb $D_f \subset \mathbb{R}$ halmazt, ahol a
	      függvény értelmezhető! Mely pontokban folytonos a függvény, miért?

	\item Határozza meg a függvény szakadási helyeit és a szakadások típusait!
	      A szakadási pontokban adja meg a függvény bal és jobb oldali
	      határértékét is!
\end{enumerate}
