\begin{enumerate}
	\item \[
		      z_1
		      = 5 \left( -\frac{1}{2} + \frac{\sqrt{3}}{2} i \right)
		      = -\frac{5}{2} + \frac{5\sqrt{3}}{2} i
	      \]

	\item $|z_2| = \sqrt{1 + 3} = 2$, és
	      $\cos \varphi = -\frac{1}{2}$, $\sin \varphi = -\frac{\sqrt{3}}{2}$,
	      így $\varphi = \frac{4\pi}{3}$:
	      \[
		      z_2
		      = 2 \left( \cos \frac{4\pi}{3} + i \sin \frac{4\pi}{3} \right)
		      = 2 e^{i \frac{4\pi}{3}}
	      \]

	\item Trigonometrikus alakban szorozva:
	      \[
		      z_1 \cdot z_2
		      = 5 \cdot 2 \left(
		      \cos\left(\frac{2\pi}{3} + \frac{4\pi}{3}\right)
		      + i \sin\left(\frac{2\pi}{3} + \frac{4\pi}{3}\right)
		      \right)
		      = 10 (\cos 2\pi + i \sin 2\pi)
		      = 10
	      \]

	\item \[
		      z_2^{10}
		      = 2^{10} e^{i \frac{40\pi}{3}}
		      = 1024 e^{i \frac{4\pi}{3}}
		      = 1024 \left( \cos \frac{4\pi}{3} + i \sin \frac{4\pi}{3} \right)
	      \]
	      A számok a komplex számsíkon ($z_2^{10}$ hossza ($1024$) miatt nem fér az ábrára; argumentuma
	      $\frac{4\pi}{3}$, ugyanaz, mint $z_2$-é.):
	      \begin{center}
	      	\begin{tikzpicture}[scale=0.4, >=stealth]
	      		\draw[->] (-5, 0) -- (12.4, 0) node[right] {$\operatorname{Re}$};
	      		\draw[->] (0, -5) -- (0, 5) node[above] {$\operatorname{Im}$};
	      		\draw (0, 0) -- (120:5);
	      		\draw (0, 0) -- (240:2);
	      		\fill (120:5) circle (2pt) node[left] {$z_1$};
	      		\fill (240:2) circle (2pt) node[left] {$z_2$};
	      		\fill (10, 0) circle (2pt) node[above] {$z_1 \cdot z_2$};
	      	\end{tikzpicture}
	      \end{center}
\end{enumerate}
