\begin{enumerate}
	\item $\lim_{x \to x_0} f(x) = A$, ha minden $\varepsilon > 0$ számhoz
	      létezik olyan $\delta(\varepsilon) > 0$, hogy
	      $0 < |x - x_0| < \delta(\varepsilon)$ esetén
	      $|f(x) - A| < \varepsilon$.

	\item Legyen $\varepsilon > 0$.
	      \[
		      \left| \sqrt{4x + 5} - 3 \right|
		      =
		      \left| \frac{4x + 5 - 9}{\sqrt{4x + 5} + 3} \right|
		      <
		      \frac{4|x - 1|}{3}
		      < \varepsilon
		      \Longleftarrow
		      |x - 1| < \frac{3\varepsilon}{4},
	      \]
	      vagyis
	      \(\displaystyle
	      \delta(\varepsilon) = \frac{3\varepsilon}{4}
	      \).
\end{enumerate}
