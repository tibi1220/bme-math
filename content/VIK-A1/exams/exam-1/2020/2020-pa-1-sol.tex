\[
	\frac{3 - 3i}{2 + i}
	= \frac{(3 - 3i)(2 - i)}{(2 + i)(2 - i)}
	= \frac{6 + 3i^2 - 3i - 6i}{4 - i^2},
\]
ezért
\(
\operatorname{Re}\left(\frac{3 - 3i}{2 + i}\right) = \frac{3}{5}
\) és
\(
\operatorname{Im}\left(\frac{3 - 3i}{2 + i}\right) = -\frac{9}{5}
\).
\[
	\operatorname{Re}\left(3 - 3i + \overline{2 + i}\right) = 5, \quad
	\operatorname{Im}\left(3 - 3i + \overline{2 + i}\right) = -4.
\]
