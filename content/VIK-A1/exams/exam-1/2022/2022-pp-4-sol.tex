Mivel $x^{2022} - x^{2021} = x^{2021} (x - 1)$ és
$x^{2022} + x^{2021} = x^{2021} (x + 1)$:
\begin{enumerate}
	\item \[
		      \lim_{x \to -\infty} f(x)
		      = \lim_{x \to -\infty}
		      \left( \frac{x}{|x|} \right)^{2022} \cdot
		      \frac{1 - \frac{1}{x}}{\left| 1 + \frac{1}{x} \right|}
		      = 1
	      \]

	\item \[
		      \lim_{x \to 0 \pm 0} f(x)
		      = \lim_{x \to 0 \pm 0}
		      \left( \frac{x}{|x|} \right)^{2021} \cdot
		      \frac{x - 1}{|x + 1|}
		      = \pm 1 \cdot (-1)
		      = \mp 1,
	      \]
	      a két oldali határérték különbözik, így $\lim_{x \to 0} f(x)$ nem
	      létezik.

	\item $f$ folytonos az $1$ pontban, így
	      \(
	      \lim_{x \to 1} f(x) = f(1) = \frac{0}{2} = 0
	      \).
\end{enumerate}
