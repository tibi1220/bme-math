Számolja ki az alábbi sorozatok határértékét!
\begin{enumerate}
	\item \(\displaystyle
	      a_n = \sqrt[n]{\frac{7^n + 5^n}{n^3 + 2}},
	      \)

	\item \(\displaystyle
	      b_n = \left(
	      \frac{2n^2 - 3}{2n^2 + 4}
	      \right)^{3n^2},
	      \)

	\item \(\displaystyle
	      c_n = \sqrt{n^3 - 5n - 3} - \sqrt{n^3 - 2n + 6}
	      \)
\end{enumerate}
