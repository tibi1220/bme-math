\begin{enumerate}
	\item \(\displaystyle
	      a_n = \frac{1}{\sqrt{n^2 + 2n} - n};
	      \)

	\item \(\displaystyle
	      b_n = \left(
	      \frac{2n - 3}{5 + 2n}
	      \right)^{n + 2};
	      \)

	\item \(\displaystyle
	      c_n = \sqrt[n]{\frac{2^n + 5}{n^2 + 1}}
	      \)
\end{enumerate}
