\begin{itemize}
	\item \[
		      3
		      \leftarrow \frac{3}{\sqrt[n]{2} \sqrt[n]{n^3}}
		      = \sqrt[n]{\frac{3^n}{n^3 + n^3}}
		      \le a_n
		      \le \sqrt[n]{\frac{3^n + 3^n}{1}}
		      = \sqrt[n]{2} \cdot 3 \to 3,
	      \]
	      így a rendőrelv miatt $a_n \to 3$.

	\item \[
		      b_n
		      = \left( \frac{1 + \frac{4{,}5}{3n}}{1 - \frac{7{,}5}{3n}} \right)^{3n}
		      \left( \frac{2n + 3}{2n - 5} \right)^3
		      \to \frac{e^{4{,}5}}{e^{-7{,}5}} \cdot 1
		      = e^{12}
	      \]
\end{itemize}
