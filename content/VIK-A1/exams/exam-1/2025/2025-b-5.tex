\begin{enumerate}
	\item Adjon példát olyan $f : \mathbb{R} \to \mathbb{R}$ függvényre, amelynek
	      a nulla kivételével minden helyen zérus a határértéke, de a nullában nem
	      létezik a határértéke!

	\item Igazolja, hogy nem létezik olyan $\mathbb{R} \to \mathbb{R}$ függvény,
	      amelynek a nullában 1 a határértéke, minden más pontban pedig nulla!
\end{enumerate}
