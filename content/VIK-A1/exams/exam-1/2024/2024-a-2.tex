\begin{enumerate}
	\item Igazolja, hogy a fent megadott rekurzív sorozat elemeire
	      $1 < a_n < 4$ teljesül!

	\item Igazolja, hogy a sorozat monoton növő!

	\item Igazolja, hogy létezik a sorozat határértéke, és adja is meg!
\end{enumerate}
