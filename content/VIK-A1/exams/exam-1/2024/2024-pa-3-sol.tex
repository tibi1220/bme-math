\begin{enumerate}
	\item \[
		      z_1
		      = 3 \left( -\frac{1}{2} - \frac{\sqrt{3}}{2} i \right)
		      = -\frac{3}{2} - \frac{3\sqrt{3}}{2} i
	      \]

	\item $|z_2| = \sqrt{1 + 3} = 2$, és
	      $\cos \varphi = -\frac{1}{2}$, $\sin \varphi = \frac{\sqrt{3}}{2}$,
	      így $\varphi = \frac{2\pi}{3}$:
	      \[
		      z_2
		      = 2 \left( \cos \frac{2\pi}{3} + i \sin \frac{2\pi}{3} \right)
		      = 2 e^{i \frac{2\pi}{3}}
	      \]

	\item Trigonometrikus alakban szorozva:
	      \[
		      z_1 \cdot z_2
		      = 3 \cdot 2 \left(
		      \cos\left(\frac{4\pi}{3} + \frac{2\pi}{3}\right)
		      + i \sin\left(\frac{4\pi}{3} + \frac{2\pi}{3}\right)
		      \right)
		      = 6 (\cos 2\pi + i \sin 2\pi)
		      = 6
	      \]

	\item \[
		      z_2^8
		      = 2^8 e^{i \frac{16\pi}{3}}
		      = 256 e^{i \frac{4\pi}{3}}
		      = 256 \left( \cos \frac{4\pi}{3} + i \sin \frac{4\pi}{3} \right)
	      \]
	      A számok a komplex számsíkon ($z_2^8$ hossza ($256$) miatt nem fér az ábrára; argumentuma
	      $\frac{4\pi}{3}$, ugyanaz, mint $z_1$-é.):
	      \begin{center}
	      	\begin{tikzpicture}[scale=0.6, >=stealth]
	      		\draw[->] (-3.2, 0) -- (7.8, 0) node[right] {$\operatorname{Re}$};
	      		\draw[->] (0, -3.2) -- (0, 2.4) node[above] {$\operatorname{Im}$};
	      		\draw (0, 0) -- (240:3);
	      		\draw (0, 0) -- (120:2);
	      		\fill (240:3) circle (2pt) node[left] {$z_1$};
	      		\fill (120:2) circle (2pt) node[left] {$z_2$};
	      		\fill (6, 0) circle (2pt) node[above] {$z_1 \cdot z_2$};
	      	\end{tikzpicture}
	      \end{center}
\end{enumerate}
