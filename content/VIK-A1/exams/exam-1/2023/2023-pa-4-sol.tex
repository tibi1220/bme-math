\begin{enumerate}
	\item $\lim_{x \to x_0} f(x) = A$, ha minden $\varepsilon > 0$ számhoz
	      létezik $\delta(\varepsilon) > 0$, melyre
	      $0 < |x - x_0| < \delta(\varepsilon)$ esetén
	      $|f(x) - A| < \varepsilon$.

	\item Legyen $\varepsilon > 0$.
	      \[
		      \left| \frac{5x - 2}{2x + 7} + 4 \right|
		      = \frac{|5x - 2 + 8x + 28|}{|2x + 7|}
		      = 13 \frac{|x + 2|}{|2x + 7|}
	      \]
	      A további becsléshez felhasználjuk, hogy $x$ a $-2$ közelében van. Ha
	      például $|x + 2| < 1$, akkor $-3 < x < -1$, és $1 < 2x + 7 < 5$.
	      Tehát
	      \[
		      13 \frac{|x + 2|}{|2x + 7|} \le 13 |x + 2| < \varepsilon,
	      \]
	      ha $|x + 2| < \frac{1}{13} \varepsilon$, így
	      \(
	      \delta(\varepsilon) = \min\left(1, \frac{1}{13} \varepsilon\right)
	      \).
\end{enumerate}
