\begin{enumerate}
	\item Mit értünk azon, hogy $\lim_{n \to \infty} a_n = +\infty$? (Adja meg a
	      definíciót!)

	\item A definíció alapján igazolja a következő állítás helyességét!
	      \[
		      \lim_{n \to \infty} \frac{n^3 - 9n}{n - 3} = +\infty
	      \]
\end{enumerate}
