\begin{enumerate}
	\item \[
		      z
		      = \frac{1 - i}{(3 + 4i) - (2 - i)}
		      = \frac{1 - i}{1 + 5i}
		      = \frac{(1 - i)(1 - 5i)}{(1 + 5i)(1 - 5i)}
		      = \frac{-4 - 6i}{26}
		      = -\frac{2}{13} - \frac{3}{13} i,
	      \]
	      tehát $\operatorname{Re} z = -\frac{2}{13}$ és
	      $\operatorname{Im} z = -\frac{3}{13}$.

	      Trigonometrikus alakban számolva
	      $2 + 2i = 2\sqrt{2}(\cos 45^\circ + i \sin 45^\circ)$ és
	      \(
	      \overline{2(\cos 15^\circ + i \sin 15^\circ)}
	      = 2(\cos(-15^\circ) + i \sin(-15^\circ))
	      \),
	      így
	      \[
		      w
		      = 4\sqrt{2}(\cos 30^\circ + i \sin 30^\circ)
		      = 4\sqrt{2} \left( \frac{\sqrt{3}}{2} + \frac{1}{2} i \right)
		      = 2\sqrt{6} + 2\sqrt{2} i,
	      \]
	      tehát $\operatorname{Re} w = 2\sqrt{6}$ és
	      $\operatorname{Im} w = 2\sqrt{2}$. A szorzásnál a hosszak
	      összeszorzódnak, a szögek összeadódnak ($45^\circ - 15^\circ = 30^\circ$):
	      \begin{center}
	      	\begin{tikzpicture}[scale=0.7, >=stealth]
	      		\draw[->] (-0.8, 0) -- (6.2, 0) node[right] {$\operatorname{Re}$};
	      		\draw[->] (0, -1.4) -- (0, 3.6) node[above] {$\operatorname{Im}$};
	      		\draw (0, 0) -- (2, 2);
	      		\draw (0, 0) -- (-15:2);
	      		\draw[thick] (0, 0) -- (30:5.657);
	      		\fill (2, 2) circle (2pt) node[above left] {$2 + 2i$};
	      		\fill (-15:2) circle (2pt) node[below right] {$2 e^{-i 15^\circ}$};
	      		\fill (30:5.657) circle (2pt) node[above] {$w$};
	      	\end{tikzpicture}
	      \end{center}

	\item $z^2 + 2z + 3i(z + 1) = z^2 + (2 + 3i) z + 3i$, így a gyökök
	      \[
		      z_{1, 2}
		      = \frac{-(2 + 3i) \pm \sqrt{(2 + 3i)^2 - 12i}}{2}
		      = \frac{-2 - 3i \pm \sqrt{-5}}{2}
		      = \frac{-2 - 3i \pm \sqrt{5} i}{2},
	      \]
	      azaz
	      \(
	      z_1 = -1 + \frac{\sqrt{5} - 3}{2} i
	      \)
	      és
	      \(
	      z_2 = -1 - \frac{\sqrt{5} + 3}{2} i
	      \).
	      A gyökök a komplex számsíkon:
	      \begin{center}
	      	\begin{tikzpicture}[scale=0.8, >=stealth]
	      		\draw[->] (-2.4, 0) -- (1.2, 0) node[right] {$\operatorname{Re}$};
	      		\draw[->] (0, -3.2) -- (0, 1.2) node[above] {$\operatorname{Im}$};
	      		\fill (-1, -0.382) circle (2pt) node[left] {$z_1$};
	      		\fill (-1, -2.618) circle (2pt) node[left] {$z_2$};
	      	\end{tikzpicture}
	      \end{center}
\end{enumerate}
