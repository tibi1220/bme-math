\begin{enumerate}
	\item $\lim_{x \to x_0} f(x) = A$, ha minden $\varepsilon > 0$ számhoz
	      létezik olyan $\delta(\varepsilon) > 0$, hogy
	      $0 < |x - x_0| < \delta(\varepsilon)$ esetén
	      $|f(x) - A| < \varepsilon$.

	\item Legyen $\varepsilon > 0$.
	      \[
		      \left| \sqrt{7x + 9} - 4 \right|
		      =
		      \left| \frac{7x + 9 - 16}{\sqrt{7x + 9} + 4} \right|
		      <
		      \frac{7|x - 1|}{4}
		      < \varepsilon
		      \Longleftarrow
		      |x - 1| < \frac{4\varepsilon}{7},
	      \]
	      vagyis
	      \(\displaystyle
	      \delta(\varepsilon) = \frac{4\varepsilon}{7}
	      \).
\end{enumerate}
