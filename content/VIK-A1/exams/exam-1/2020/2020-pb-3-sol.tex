\begin{enumerate}
	\item \[
		      4
		      \leftarrow \frac{4}{\sqrt[n]{2} \sqrt[n]{n^4}}
		      = \sqrt[n]{\frac{4^n}{n^4 + n^4}}
		      \le a_n
		      \le \sqrt[n]{\frac{4^n + 4^n}{1}}
		      = \sqrt[n]{2} \cdot 4 \to 4,
	      \]
	      így a rendőrelv miatt $a_n \to 4$.

	\item \[
		      b_n
		      = \left( \frac{1 + \frac{0{,}8}{2n}}{1 - \frac{1{,}2}{2n}} \right)^{2n}
		      \left( \frac{5n + 2}{5n - 3} \right)^2
		      \to \frac{e^{0{,}8}}{e^{-1{,}2}} \cdot 1
		      = e^2
	      \]
\end{enumerate}
