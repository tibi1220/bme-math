Számolja ki az alábbi sorozatok határértékét:
\begin{enumerate}
	\item \(\displaystyle
	      a_n = \sqrt[n]{\frac{n^3 + 3n}{3n^2 + 5}},
	      \)

	\item \(\displaystyle
	      b_n = \sqrt{4n^2 + 3n - 4} - 2n,
	      \)

	\item \(\displaystyle
	      c_n = \left(
	      \frac{3n - 1}{3n + 2}
	      \right)^{2n}.
	      \)
\end{enumerate}
