\[
	\frac{z - i}{\overline{z} - i}
	= \frac{\sqrt{12} - 3i}{\sqrt{12} + i}
	= \frac{(\sqrt{12} - 3i)(\sqrt{12} - i)}{(\sqrt{12} + i)(\sqrt{12} - i)}
	= \frac{12 - 3 - 4\sqrt{12} i}{13}
\]
\[
	\operatorname{Re}\left(\frac{z - i}{\overline{z} - i}\right) = \frac{9}{13}
	\quad \text{és} \quad
	\operatorname{Im}\left(\frac{z - i}{\overline{z} - i}\right) = -\frac{8\sqrt{3}}{13}.
\]

$|z| = 4$, $\arg z = -30^\circ$, vagyis
$z = 4\left(\cos(-30^\circ) + i \sin(-30^\circ)\right)$.
\[
	\begin{aligned}
		\sqrt{iz}
		 & = \sqrt{(\cos 90^\circ + i \sin 90^\circ) \cdot 4\left(\cos(-30^\circ) + i \sin(-30^\circ)\right)}
		= \sqrt{4(\cos 60^\circ + i \sin 60^\circ)}                                                  \\
		 & = 2\left(\cos(30^\circ + k \cdot 180^\circ) + i \sin(30^\circ + k \cdot 180^\circ)\right),
		\quad k = 0, 1
	\end{aligned}
\]
\[
	\operatorname{Re}\left(\sqrt{iz}\right) = \pm \sqrt{3}, \quad
	\operatorname{Im}\left(\sqrt{iz}\right) = \pm 1.
