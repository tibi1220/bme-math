\begin{enumerate}
	\item $\displaystyle\lim_{n \to \infty} a_n = A$, ha $\forall\varepsilon > 0$
	      esetén $\exists N(\varepsilon) \in \mathbb{N}$, hogy
	      $|a_n - A| < \varepsilon$, ha $n > N(\varepsilon)$.


	\item Helyettesítsünk be a definícióban szereplő egyenlőtlenségbe:
	      \[
		      \left| \frac{6n^2 + n + 1}{3n^2 + 30} - 2 \right|
		      = \left| \frac{6n^2 + n + 1 - 6n^2 - 60}{3n^2 + 30} \right|
		      = \frac{|n - 59|}{3n^2 + 30}
		      \text.
	      \]
	      $|n - 59| = n - 59$, ha $n \ge 59$, ekkor:
	      \[
		      \frac{|n - 59|}{3n^2 + 30}
		      = \frac{n - 59}{3n^2 + 30}
		      \text.
	      \]
	      Becsüljük felül a törtünket. Ezt a számláló növelésével, illetve a
	      nevező csökkentésével tudjuk elérni:
	      \[
		      \frac{n - 59}{3n^2 + 30}
		      \le \frac{n}{3n^2}
		      = \frac{1}{3n}
		      < \varepsilon
		      \qquad \Rightarrow \qquad
		      n > \frac{1}{3 \varepsilon}
		      \text.
	      \]
	      A megoldás tehát:
	      \[
		      N(\varepsilon) > \max \left\{ \frac{1}{3\varepsilon}; 59 \right\}
		      \text, \qquad \text{vagy} \qquad
		      N(\varepsilon) = \left[
			      \max \left\{ \frac{1}{3\varepsilon}; 59 \right\}
			      \right] + 1
		      \text.
	      \]
\end{enumerate}
